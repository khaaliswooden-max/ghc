\section{The Halal-Closure Functor}
\label{sec:hcf}

\subsection{The supply-chain category $\SC$}

\begin{definition}[Supply-chain category]
\label{def:sc}
$\SC$ is the symmetric monoidal category whose
\begin{itemize}
  \item \textbf{objects} are batches: triples $(I, q, \mu)$ where $I$ is
        an ingredient identifier, $q \in \mathbb{R}_{\geq 0}$ is a
        quantity, and $\mu$ is a provenance witness;
  \item \textbf{morphisms} $f \colon B_1 \to B_2$ are processing steps
        (slaughter, mixing, separation, packaging, transport, audit),
        each carrying a $\sigma$-algebra of admissible operational
        constraints;
  \item \textbf{tensor product} $\otimes$ is parallel composition
        (independent processing of co-located batches);
  \item \textbf{unit} $I$ is the empty batch.
\end{itemize}
\end{definition}

\subsection{The compliance lattice $\Lat$}

\begin{definition}[Compliance lattice]
\label{def:lat}
$\Lat$ is the complete distributive lattice
$\{\halal, \mashbuh, \haram\}$ with $\halal \prec \mashbuh \prec \haram$,
augmented with an authority parameter $A$ so that
$\Lat = \prod_{A \in \mathcal{A}} \Lat_A$ for a (federated) authority
set $\mathcal{A}$. Meet $\sqcap$ models conjunction of authoritative
opinion; join $\sqcup$ models disjunction.
\end{definition}

\begin{theorem}[Distributive bounded lattice; mechanized]
\label{thm:lattice-laws}
$(\Lat, \sqcap, \sqcup)$ is a bounded distributive lattice with
bottom $\halal$ and top $\haram$. Commutativity, associativity,
idempotence, both absorption laws, and both distributive laws hold.
\end{theorem}

\begin{proof}[Proof (kernel)]
By exhaustive case analysis on the three-element carrier; mechanized
in \texttt{proofs/Ghc/Lattice.lean} via
\texttt{Verdict.meet\_comm}, \texttt{Verdict.meet\_assoc},
\texttt{Verdict.meet\_idem}, \texttt{Verdict.absorb\_meet\_join},
\texttt{Verdict.absorb\_join\_meet},
\texttt{Verdict.meet\_join\_distrib},
\texttt{Verdict.join\_meet\_distrib},
\texttt{Verdict.halal\_bot}, \texttt{Verdict.haram\_top}.
\end{proof}

\subsection{The functor (kernel form)}

\begin{theorem}[Halal-Closure Functor — kernel]
\label{thm:hcf}
Let $\Process$ be the free monoid of supply-chain operations (lists
of per-step verdicts), and define
$\HCF \colon \Process \to \Lat$ by iterated meet of per-step
verdicts. Then
\begin{enumerate}[label=(H\arabic*)]
  \item \emph{Compositionality:}
        $\HCF(\mathrm{xs} \mathbin{+\!\!+} \mathrm{ys})
         = \HCF(\mathrm{xs}) \sqcap \HCF(\mathrm{ys})$.
  \item \emph{Tensor preservation:} on parallel-composed processes
        (independently resolved verdicts) the same identity holds.
  \item \emph{All-halal closure:} if every step verdict is $\halal$,
        $\HCF(\mathrm{xs}) = \halal$.
  \item \emph{Haram absorption:} a single $\haram$ step taints the
        entire process.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof (kernel)]
By induction on the operation list; mechanized in
\texttt{proofs/Ghc/Cate\-gory.lean} via
\texttt{Process.halalClosure\_append} (compositionality),
\texttt{Process.halal\-Closure\_parallel} (tensor preservation),
\texttt{Process.halalClosure\_all\_halal}, and
\texttt{Process.halalClosure\_haram\_absorb}.
\end{proof}

\begin{remark}[Phase B+ extension]
The extension to a genuine \texttt{MonoidalCategory} structure ---
which carries the full strong-monoidal coherence required by
\cite{maclane,fongspivak} --- introduces no new compositional content
beyond the kernel; it only re-encodes the same identities at the
level of \texttt{mathlib}'s \texttt{Functor} class. We track this on
the Phase B+ roadmap.
\end{remark}

\begin{remark}
The compositionality clause means a verifier can recompute compliance
status of a finished product from any honest intermediate evidence,
which is what makes the entire downstream pipeline (EPCIS event
capture $\to$ zk-Halal attestation) sound.
\end{remark}
